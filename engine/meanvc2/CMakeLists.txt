cmake_minimum_required(VERSION 3.20)
project(ezlocalai_meanvc2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EZLOCALAI_AUDIOCPP_SOURCE)
    message(FATAL_ERROR "Set EZLOCALAI_AUDIOCPP_SOURCE to the pinned audio.cpp checkout")
endif()
# Apply once, or accept exactly the already-applied patch. Refuse drift.
execute_process(COMMAND git -C "${EZLOCALAI_AUDIOCPP_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT revision STREQUAL "cd770076152085af5b166521185bec76f9d47808")
    message(FATAL_ERROR "Unexpected audio.cpp source revision")
endif()
set(speaker_patch "${CMAKE_CURRENT_SOURCE_DIR}/speaker-cache.patch")
execute_process(COMMAND git -C "${EZLOCALAI_AUDIOCPP_SOURCE}" apply --check "${speaker_patch}"
    RESULT_VARIABLE apply_result OUTPUT_QUIET ERROR_QUIET)
if(apply_result EQUAL 0)
    execute_process(COMMAND git -C "${EZLOCALAI_AUDIOCPP_SOURCE}" apply "${speaker_patch}" COMMAND_ERROR_IS_FATAL ANY)
else()
    execute_process(COMMAND git -C "${EZLOCALAI_AUDIOCPP_SOURCE}" apply --reverse --check "${speaker_patch}"
        RESULT_VARIABLE reverse_result OUTPUT_QUIET ERROR_QUIET)
    if(NOT reverse_result EQUAL 0)
        message(FATAL_ERROR "MeanVC2 speaker cache patch does not apply cleanly")
    endif()
endif()
file(SHA256 "${EZLOCALAI_AUDIOCPP_SOURCE}/include/engine/models/meanvc2/session.h" source_hash)
if(NOT source_hash STREQUAL "b1c7d3665d7204b4160aa7b4b4dfae4c7336de01ed4810a9a4f3a29f5d44b5de")
    message(FATAL_ERROR "Unexpected patched MeanVC2 source: include/engine/models/meanvc2/session.h")
endif()
file(SHA256 "${EZLOCALAI_AUDIOCPP_SOURCE}/src/models/meanvc2/session.cpp" source_hash)
if(NOT source_hash STREQUAL "ebd494d37d9ae85aae380592e34664974535024977f42c61312bde637da98a9a")
    message(FATAL_ERROR "Unexpected patched MeanVC2 source: src/models/meanvc2/session.cpp")
endif()
set(AUDIOCPP_MODEL_SET custom CACHE STRING "" FORCE)
set(AUDIOCPP_MODELS meanvc2 CACHE STRING "" FORCE)
set(AUDIOCPP_DEPLOYMENT_BUILD ON CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_NATIVE_CPU OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_METAL OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_VULKAN OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(ENGINE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENGINE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(AUDIOCPP_BUILD_NATIVE_MODEL_MANAGER OFF CACHE BOOL "" FORCE)
add_subdirectory("${EZLOCALAI_AUDIOCPP_SOURCE}" audiocpp EXCLUDE_FROM_ALL)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(ezlocalai-meanvc2 worker.cpp)
set_target_properties(ezlocalai-meanvc2 PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_include_directories(ezlocalai-meanvc2 PRIVATE ../../include)
target_link_libraries(ezlocalai-meanvc2 PRIVATE engine_runtime nlohmann_json::nlohmann_json OpenSSL::Crypto)
install(TARGETS ezlocalai-meanvc2 RUNTIME DESTINATION bin)

option(EZLOCALAI_MEANVC2_VALIDATION "Build the real-model conditioning reuse probe" OFF)
if(EZLOCALAI_MEANVC2_VALIDATION)
    add_executable(meanvc2-session-validation session_validation.cpp)
    target_include_directories(meanvc2-session-validation PRIVATE ../../include)
    target_link_libraries(meanvc2-session-validation PRIVATE engine_runtime nlohmann_json::nlohmann_json)
    set_target_properties(meanvc2-session-validation PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
